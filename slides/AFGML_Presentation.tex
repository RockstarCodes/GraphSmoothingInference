\begin{frame}{Research question and decision context}
\begin{itemize}
\item Offline audit of node classification on Pokec-z (FairGNN release); not a recruitment system.
\item $Y=1$: benchmark's positive working-field class (raw code $\ge1$, meaning undocumented). $D=1$: pipeline predicts that class.
\item \textbf{Question:} when does probabilistic graph smoothing undo the fairness gains of debiased embeddings, and can constrained edge potentials reduce this effect?
\item \textbf{Hypothesis:} with demographic homophily + base-rate gap, uniform smoothing widens the gender TPR gap ($\Delta$EOpp); falsified if paired block-bootstrap effects are $\le 0$ / not significant after Holm.
\end{itemize}
\end{frame}

\begin{frame}{Closest work and proposed contribution}
\begin{itemize}
\item \textbf{Established:} FairWalk debiases walks (Rahman et al.\ 2019, reproduced); FairDrop/CrossWalk reweight homophilous edges during representation learning; Correct\&Smooth smooths a base predictor, no fairness analysis; FairPSL: fairness in relational MAP inference.
\item \textbf{Open (within 12 verified papers):} fairness change \emph{between} a fixed debiased embedding and a downstream exact CMN, under controlled homophily.
\item \textbf{Contribution:} exact-inference decomposition + homophily-constrained attenuation (HCA) $\lambda_{ij}=\lambda(1-\eta\,\mathbf 1[A_i{=}A_j])$.
\item \textbf{Evidence needed:} HCA must beat a \emph{global scalar} with the same tuning budget (H4); otherwise it is just ``smooth less''.
\end{itemize}
\end{frame}

\end{tikzpicture}\\[2pt]
{\scriptsize grey = observed training label; squares = unary $\psi_i(Y_i;H_i)$; edges = $\psi_{ij}$ on forest $E_F$}
\column{0.58\textwidth}
\[
P(\mathbf Y\!\mid\!\mathbf H,E_F)=\tfrac1Z\prod_i\psi_i(Y_i)\!\!\prod_{(i,j)\in E_F}\!\!e^{\lambda_{ij}\mathbf 1[Y_i=Y_j]}
\]
\begin{itemize}\small
\item Query: $q_i=P(Y_i{=}1\mid\mathbf y_{\mathrm{train}})$, exact log-space sum-product on max-Jaccard forest (\RForestKept\% of edges).
\item Markov blanket = forest neighbours; each positive neighbour adds $\lambda_{ij}$ to log-odds.
\item Validation: enumeration = sum-product = VE = clique tree, max error \RInfMaxErr{} on \RInfFixtures{} fixtures (tol.\ $10^{-8}$).
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Fairness criterion and proposed method}
\begin{itemize}
\item \textbf{Primary:} gender $\Delta$EOpp $=\max_g\mathrm{TPR}_g-\min_g\mathrm{TPR}_g$: base rates of the recorded label differ, the risk is unequal errors among true positives. Diagnostics: $\Delta$DP, $\Delta$EOdds, $R_{sel}$, intersectional, calibration.
\item \textbf{Utility:} macro-F1; acceptable loss 0.02 (validation).
\item \textbf{HCA:} $\lambda$ fit on training folds (held-out log-likelihood); $\eta$ = argmin val $\Delta$EOpp s.t.\ val F1 $\ge$ uniform $-0.02$.
\item \textbf{Protected attribute at prediction time:} HCA yes (edge weights); FairWalk only for walks; group thresholds yes.
\end{itemize}
\end{frame}

\begin{frame}{Experimental protocol}
\begin{itemize}
\item Baselines: feature LR, SGC, Node2Vec+LR, FairWalk+LR, group thresholds, uniform CMN; ablations: no pairwise, global scalar, Node2Vec for FairWalk.
\item Same graph, splits, forest, threshold rule and 11-point grids for every method.
\item 5 seeds (training variability) \textbf{separately from} 1{,}000 Louvain-block bootstrap replicates (\RNBlocks{} blocks; evaluation uncertainty) and 10 synthetic realisations/cell.
\item Confirmatory H1--H4 (Holm), declared before test evaluation; all else exploratory.
\item Exp.~4: simulated homophily $\times$ degree imbalance; Exp.~5 + robustness: dropout, corruption, rewiring, 4 forests, 3 dims, 4 label budgets.
\end{itemize}
\end{frame}

\begin{frame}{Ablation, robustness and failure case}
\begin{itemize}
\item \textbf{Ablation A:} no pairwise factors reproduces unary predictions exactly ($\Delta$ = 0).
\item \textbf{Ablation B:} HCA vs global scalar \RHfourEst{} [\RHfourLo, \RHfourHi] --- no evidence group-aware attenuation helps.
\item \textbf{Ablation C:} Node2Vec HCA$-$uniform \REthreeDeoppG{} [\REthreeDeoppGLo, \REthreeDeoppGHi].
\item \textbf{Robustness} (\RRobConditions{} conditions): smoothing effect on $\Delta$EOpp $-0.027$ to $0.019$ (FairWalk), no consistent sign; F1 gain needs the Jaccard forest.
\item \textbf{Failure:} validation-selected $\eta$ unstable across seeds (\REtaFwList); weak fitted coupling $\lambda=\RLambdaValues$.
\item Decision thresholds: $\Delta$EOpp \RNVThrDeoppG{} but $\Delta$DP \RNVThrDdpG{} --- criteria conflict.
\end{itemize}
\end{frame}

\begin{frame}{Conclusion and reproducibility}
\begin{itemize}
\item \textbf{Conclusion:} smoothing reintroduces bias only when neighbourhoods carry group base rates; not the case for gender on this Pokec-z subgraph. HCA gives no gain over weaker smoothing.
\item \textbf{Limits/ethics:} one subgraph; region leaks from embeddings (AUROC \RSensNVRegion); HCA is differential edge treatment; label has no employment use.
\item \textbf{Reproduce:} \texttt{scripts/run\_experiments.py}, \texttt{evaluate.py}, \texttt{make\_results.py}; commit [hash].
\item AARON: [contribution and verification]. AASHISH: [contribution and verification].
\end{itemize}
\end{frame}
